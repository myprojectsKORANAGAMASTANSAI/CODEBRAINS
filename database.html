Create Folders Table create table if not exists cb_folders ( id text primary key, parent_id text default 'root', name text not null, category text default 'General', date text, created_at timestamp with time zone default timezone('utc'::text, now()) ); -- 2. Create Files Table create table if not exists cb_files ( id text primary key, folder_id text default 'root', name text not null, type text default 'code', size text, date text, content text, created_at timestamp with time zone default timezone('utc'::text, now()) ); -- 3. Enable Public Access & Realtime Policies alter table cb_folders enable row level security; alter table cb_files enable row level security; create policy "Allow all on cb_folders" on cb_folders for all using (true) with check (true); create policy "Allow all on cb_files" on cb_files for all using (true) with check (true); alter publication supabase_realtime add table cb_folders; alter publication supabase_realtime add table cb_files;
